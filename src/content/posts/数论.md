## 快速幂

> 在幂指数过大的情况下，快速幂运算

```c
int power(int a, int n) {
  int tp = 1;
  while (n) {
    if (n & 1) tp = 1ll * tp * a % mod;
    a = 1ll * a * a % mod, n >>= 1;
  }
  return tp;
}
```

### 分治

> 将$a^{100}$化成$a^{50}*a^{50}$,递归处理

```c
#include <bits/stdc++.h>
using namespace std;

typedef long long ll;

ll a, b, m;
int getRes(ll a, ll b, ll m) {
	if (b == 0) return 1;
	if (b % 2 == 1) {
		return a * getRes(a, b - 1, m) % m;
	} else {
		ll mul = getRes(a, b / 2, m);
		return mul * mul % m;
	}
}

int main() {
	cin >> a >> b >> m;
	cout << getRes(a, b, m); 
	return 0;
}
```

### 迭代

> 将幂指数作二进制数位运算

```c
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

ll a, b, m;

int main() {
	cin >> a >> b >> m;
	ll ans = 1, cur = a;
	while (b > 0) {
		if (b & 1) ans = ans * cur % m;
		cur = cur * cur % m;
		b >>= 1;
	}
	cout << ans;
	return 0;
}
```
